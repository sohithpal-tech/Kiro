def calc_ema(series,period): return series.ewm(span=period,adjust=False).mean()

def calc_rsi(series,period=14):
    delta=series.diff(); gain=delta.clip(lower=0); loss=(-delta).clip(lower=0)
    ag=gain.ewm(com=period-1,adjust=False).mean(); al=loss.ewm(com=period-1,adjust=False).mean()
    return 100-(100/(1+ag/al.replace(0,np.nan)))

def calc_atr(df,period=14):
    hl=df['high']-df['low']
    hc=(df['high']-df['close'].shift(1)).abs()
    lc=(df['low']-df['close'].shift(1)).abs()
    return pd.concat([hl,hc,lc],axis=1).max(axis=1).rolling(period).mean()

def calc_adx(df,period=14):
    """ADX using correct Wilder RMA: ewm(com=period-1) — Fix A1"""
    high,low,close=df['high'],df['low'],df['close']
    tr=pd.concat([high-low,(high-close.shift(1)).abs(),(low-close.shift(1)).abs()],axis=1).max(axis=1)
    up=high.diff(); down=-low.diff()
    dm_p=np.where((up>down)&(up>0),up,0.0)
    dm_m=np.where((down>up)&(down>0),down,0.0)
    # A1 FIX: use com=period-1 for all smoothing (correct Wilder RMA)
    atr_s=pd.Series(tr).ewm(com=period-1,adjust=False).mean()
    di_p=100*pd.Series(dm_p).ewm(com=period-1,adjust=False).mean()/atr_s.replace(0,np.nan)
    di_m=100*pd.Series(dm_m).ewm(com=period-1,adjust=False).mean()/atr_s.replace(0,np.nan)
    dx=(100*(di_p-di_m).abs()/(di_p+di_m).replace(0,np.nan))
    adx=dx.ewm(com=period-1,adjust=False).mean()
    return adx

def get_trend_1h(df_1h):
    df=df_1h.copy(); df['ema50']=calc_ema(df['close'],50); df['ema200']=calc_ema(df['close'],200)
    slope=df['ema50'].iloc[-1]-df['ema50'].iloc[-5]; last=df.iloc[-1]
    if last['ema50']>last['ema200'] and slope>0: return "BUY"
    elif last['ema50']<last['ema200'] and slope<0: return "SELL"
    return None

def detect_liquidity_sweep(df,lookback=20):
    """B3 FIX: requires body/range >= 30% to filter random wicks."""
    if len(df)<lookback+2: return None
    last=df.iloc[-1]
    candle_range=last['high']-last['low']
    body_size=abs(last['close']-last['open'])
    if candle_range==0 or body_size/candle_range<0.30: return None
    prev_low=df['low'].iloc[-lookback-1:-1].min()
    prev_high=df['high'].iloc[-lookback-1:-1].max()
    if last['low']<prev_low and last['close']>prev_low: return "BUY"
    if last['high']>prev_high and last['close']<prev_high: return "SELL"
    return None

def detect_mss(df,sweep_direction):
    """B4 FIX: requires candle body >= 40% of range for decisive confirmation."""
    if len(df)<12: return False
    last=df.iloc[-1]
    candle_range=last['high']-last['low']
    body_size=abs(last['close']-last['open'])
    if candle_range==0 or body_size/candle_range<0.40: return False
    if sweep_direction=="BUY": return last['close']>df['high'].iloc[-11:-1].max()
    if sweep_direction=="SELL": return last['close']<df['low'].iloc[-11:-1].min()
    return False

def detect_fvg(df,direction):
    """B2 FIX: checks current price is still outside the gap (not already filled)."""
    if len(df)<4: return None
    c0,c1,c2=df.iloc[-1],df.iloc[-2],df.iloc[-3]
    current_price=c0['close']
    if direction=="BUY" and c2['high']<c0['low']:
        gap_low,gap_high=c2['high'],c0['low']
        if current_price>=gap_low: return (gap_low,gap_high)
    if direction=="SELL" and c2['low']>c0['high']:
        gap_low,gap_high=c0['high'],c2['low']
        if current_price<=gap_high: return (gap_low,gap_high)
    return None

def generate_signal(df_15m,df_1h,df_4h,regime):
    """C1: 4H bias filter integrated. Blocks signals against 4H EMA trend."""
    df_15m['rsi']=calc_rsi(df_15m['close']); rsi=df_15m['rsi'].iloc[-1]
    if "TREND" in regime:
        trend_dir=get_trend_1h(df_1h)
        if trend_dir is None: return None
        sweep=detect_liquidity_sweep(df_15m)
        if sweep is None or sweep!=trend_dir: return None
        if not detect_mss(df_15m,sweep): return None
        if detect_fvg(df_15m,sweep) is None: return None
        if sweep=="BUY" and rsi>65: return None
        if sweep=="SELL" and rsi<35: return None
        # C1 FIX: 4H bias filter
        if df_4h is not None:
            bias_4h=get_4h_bias(df_4h)
            if bias_4h is not None and sweep!=bias_4h: return None
        return sweep
    elif regime=="RANGE" and get_mode("range_allowed"):
        if rsi<30: return "BUY"
        elif rsi>70: return "SELL"
    return None

def build_ml_features(signal,atr,sl_dist,regime,rsi_val=50.0):
    """A2 FIX: rr_quality is now dynamic. rsi_at_entry added as new feature."""
    now=datetime.utcnow(); hour=now.hour; dow=now.weekday()
    session_enc=0 if 6<=hour<12 else 1 if 12<=hour<18 else 2
    sig_enc=1 if signal=="BUY" else 0
    # A2 FIX: dynamic R:R calculation
    tp_dist=sl_dist*2.0
    rr=round(tp_dist/sl_dist,3) if sl_dist>0 else 2.0
    sl_atr=round(sl_dist/atr,3) if atr>0 else 1.0
    atr_tier=0 if atr<1.5 else 1 if atr<3.0 else 2
    regime_enc={"TREND_BULL":0,"TREND_BEAR":1,"RANGE":2,"VOLATILE":3,"UNKNOWN":4}.get(regime,4)

    return pd.DataFrame([{
        "sl_distance":sl_dist,"rr_quality":rr,"atr_used":atr,
        "sl_atr_ratio":sl_atr,"lot":lot,"hour":hour,"day_of_week":dow,
        "session_enc":session_enc,"signal_enc":sig_enc,
        "atr_tier":atr_tier,"regime_enc":regime_enc,
        "rsi_at_entry":round(rsi_val,2),   # A2: new feature

def ml_filter(signal,atr,sl_dist,regime,rsi_val=50.0):
    threshold=get_mode("ml_threshold")
    if ml_xgb is None and ml_rf is None and ml_lr is None: return True
    features=build_ml_features(signal,atr,sl_dist,regime,rsi_val)
    scores,weights=[],[]
    try:
        if ml_xgb: scores.append(ml_xgb.predict_proba(features)[0][1]); weights.append(0.4)
    except Exception as e: log.warning(f"XGBoost: {e}")
    try:
        if ml_rf: scores.append(ml_rf.predict_proba(features)[0][1]); weights.append(0.3)
    except Exception as e: log.warning(f"RF: {e}")
    try:
        if ml_lr: scores.append(ml_lr.predict_proba(features)[0][1]); weights.append(0.3)
    except Exception as e: log.warning(f"LR: {e}")
    if not scores: return True
    total_w=sum(weights); final=sum(s*w for s,w in zip(scores,weights))/total_w
    log.info(f"ML score: {round(final*100,1)}% (threshold: {int(threshold*100)}%)")
    if final<threshold:
        telegram(f"🤖 ML rejected | Score: {round(final*100,1)}% < {int(threshold*100)}%")
        return False
    return True

Common Beginner Mistakes — And How to Fix Them
Mistake 1 — Forgot to tick "Add Python to PATH" during installation
What happens: When you type python --version in CMD you get an error like "'python' is not recognized as an internal or external command".

How to fix: Go to Control Panel → Add or Remove Programs → find Python 3.12 → Uninstall it. Then go back to python.org, download the installer again, and this time on the very first screen of the installer, tick the "Add Python to PATH" checkbox before clicking Install Now.
Mistake 2 — MT5 is closed when running the bot
What happens: The bot starts but immediately shows errors like "MT5 init failed" or "MT5 connection failed after retries" in CMD, and you receive a Telegram message saying "MT5 connection failed".

How to fix: Always open MetaTrader 5 first and make sure it is fully logged in and showing XAUUSD price data before you run python bot.py. The bot connects to the running MT5 app — it cannot work without it.
Mistake 3 — Wrong MT5 server name in bot.py
What happens: MT5 is open but the bot still cannot connect. You see "MT5 init failed" or an authentication error.

How to find the correct server name: Open MetaTrader 5 → click File → Login to Trade Account → look at the "Server" dropdown — the exact text shown there is what you must enter as MT5_SERVER in bot.py. It is usually something like "Exness-MT5Trial" or "Exness-Demo". Copy it exactly including the hyphen and capitalisation.
Mistake 4 — Telegram token has spaces or is incomplete
What happens: Bot runs but you never receive any Telegram messages. No errors appear in CMD.

How to check: Open bot.py in Notepad. Look at the TELEGRAM_TOKEN line. The token must be inside quote marks and have no spaces — it should look like: TELEGRAM_TOKEN = "7123456789:ABCdefGHIjkl...". The full token is about 45 characters long. If it looks shorter, you may have accidentally truncated it when copying. Go back to @BotFather in Telegram, type /mybots, select your bot, click API Token, and copy the full token again.
Mistake 5 — Bot runs but no Telegram message received
Step-by-step debug:
1. Check that your CHAT_ID in bot.py is your personal chat ID from @userinfobot — not your phone number and not your Telegram username.
2. Make sure you have started a conversation with your bot — search for your bot's username in Telegram and press Start. If you never pressed Start, the bot cannot send you messages.
3. Check that TELEGRAM_TOKEN and CHAT_ID are both inside quote marks in bot.py.
4. In CMD, run the health check: curl http://localhost:5000/health. If this returns a response, the bot is alive. The Telegram issue is credential-related, not a bot crash.
Mistake 6 — "pip is not recognized" when trying to install libraries
What happens: You type the pip install command and get "'pip' is not recognized".

How to fix: This is the same root cause as Mistake 1 — Python PATH was not set. Fix it the same way: uninstall Python, reinstall with the PATH checkbox ticked. Alternatively, try replacing pip with python -m pip in the install command — this sometimes works even without PATH being set correctly.

def calc_ema(s,p): return s.ewm(span=p,adjust=False).mean()
def calc_rsi(s,p=14):
    d=s.diff();g=d.clip(lower=0);l=(-d).clip(lower=0)
    return 100-(100/(1+g.ewm(com=p-1,adjust=False).mean()/l.ewm(com=p-1,adjust=False).mean().replace(0,np.nan)))
def calc_atr(df,p=14):
    tr=pd.concat([df['high']-df['low'],(df['high']-df['close'].shift(1)).abs(),
                  (df['low']-df['close'].shift(1)).abs()],axis=1).max(axis=1)
    return tr.rolling(p).mean()
def calc_adx(df,p=14):
    high,low,close=df['high'],df['low'],df['close']
    tr=pd.concat([high-low,(high-close.shift(1)).abs(),(low-close.shift(1)).abs()],axis=1).max(axis=1)
    up=high.diff();down=-low.diff()
    dm_p=np.where((up>down)&(up>0),up,0.0);dm_m=np.where((down>up)&(down>0),down,0.0)
    atr_s=pd.Series(tr).ewm(com=p-1,adjust=False).mean()
    di_p=100*pd.Series(dm_p).ewm(com=p-1,adjust=False).mean()/atr_s.replace(0,np.nan)
    di_m=100*pd.Series(dm_m).ewm(com=p-1,adjust=False).mean()/atr_s.replace(0,np.nan)
    return (100*(di_p-di_m).abs()/(di_p+di_m).replace(0,np.nan)).ewm(com=p-1,adjust=False).mean()

    if r==0 or b/r<0.30: return None

    if last['low']<prev_low and last['close']>prev_low: return "BUY"

def simulate_tiered_exit(df_future,signal,entry_price,sl,atr,lot):
    sl_distance=abs(entry_price-sl)

    lot1=round(lot*0.30,2);lot2=round(lot*0.30,2)

    trail_sl=sl;tier1_done=False;tier2_done=False
    total_pnl=0.0
    info=mt5.symbol_info(SYMBOL)
    spread_cost=(SPREAD_COST.get("TREND_BULL",30)*info.point*lot*100) if info else 0
    slippage_cost=(SLIPPAGE_POINTS*info.point*lot*100) if info else 0
    for i,row in df_future.iterrows():
        high=row['high'];low=row['low']
        if signal=="BUY":
            if low<=trail_sl:
                total_pnl+=(trail_sl-entry_price)*lots_remaining*100
                return {"pnl":round(total_pnl-spread_cost-slippage_cost,2),
                        "exit_reason":"SL","bars_held":i+1,
                        "r_multiple":round(total_pnl/(sl_distance*lot*100),2) if sl_distance>0 else 0}
            if not tier1_done and high>=entry_price+tp1:
                total_pnl+=tp1*lot1*100;lots_remaining-=lot1;tier1_done=True;trail_sl=entry_price
            if tier1_done and not tier2_done and high>=entry_price+tp2:
                total_pnl+=tp2*lot2*100;lots_remaining-=lot2;tier2_done=True;trail_sl=entry_price+tp1
            if tier2_done:
                new_t=high-atr*1.0
                if new_t>trail_sl: trail_sl=new_t
        elif signal=="SELL":
            if high>=trail_sl:
                total_pnl+=(entry_price-trail_sl)*lots_remaining*100
                return {"pnl":round(total_pnl-spread_cost-slippage_cost,2),
                        "exit_reason":"SL","bars_held":i+1,
                        "r_multiple":round(total_pnl/(sl_distance*lot*100),2) if sl_distance>0 else 0}
            if not tier1_done and low<=entry_price-tp1:
                total_pnl+=tp1*lot1*100;lots_remaining-=lot1;tier1_done=True;trail_sl=entry_price
            if tier1_done and not tier2_done and low<=entry_price-tp2:
                total_pnl+=tp2*lot2*100;lots_remaining-=lot2;tier2_done=True;trail_sl=entry_price-tp1

    last_price=df_future['close'].iloc[-1]
